cmake_minimum_required(VERSION 3.16)
project(coupled_handover LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)

add_library(coupled_handover SHARED src/controllers.cpp)
target_compile_features(coupled_handover PRIVATE cxx_std_20)
target_compile_options(coupled_handover PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(coupled_handover controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(controller_interface controllers.xml)
install(TARGETS coupled_handover LIBRARY DESTINATION lib)
install(PROGRAMS scripts/recover_pair DESTINATION lib/${PROJECT_NAME})
ament_package()
