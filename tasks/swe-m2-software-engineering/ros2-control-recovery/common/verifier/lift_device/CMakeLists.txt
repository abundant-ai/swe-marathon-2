cmake_minimum_required(VERSION 3.16)
project(lift_device LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)

add_library(lift_device SHARED src/device.cpp)
target_compile_features(lift_device PRIVATE cxx_std_20)
target_compile_options(lift_device PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(lift_device hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface hardware.xml)
install(TARGETS lift_device LIBRARY DESTINATION lib)
ament_package()
